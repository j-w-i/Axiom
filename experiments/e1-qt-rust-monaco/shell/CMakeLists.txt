cmake_minimum_required(VERSION 3.21)
project(e1_shell LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Core Gui Qml Quick WebEngineQuick WebChannel)
qt_add_executable(e1_shell main.cpp)
qt_add_qml_module(e1_shell URI E1 VERSION 1.0 QML_FILES Main.qml)
target_link_libraries(e1_shell PRIVATE Qt6::Core Qt6::Gui Qt6::Qml Qt6::Quick Qt6::WebEngineQuick Qt6::WebChannel)
